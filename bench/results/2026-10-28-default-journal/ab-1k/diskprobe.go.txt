package main

import (
	"fmt"
	"os"
	"sort"
	"syscall"
	"time"
)

func main() {
	f, _ := os.OpenFile("/tmp/freshdb-bench/diskp.bin", os.O_RDWR|os.O_CREATE|os.O_TRUNC, 0o644)
	z := make([]byte, 64<<20)
	f.WriteAt(z, 0)
	f.Sync()
	b := make([]byte, 1302)
	var lat []time.Duration
	begin := time.Now()
	off := int64(0)
	for i := 0; i < 8000; i++ {
		next := begin.Add(time.Duration(i) * time.Millisecond)
		time.Sleep(time.Until(next))
		t0 := time.Now()
		f.WriteAt(b, off)
		off += int64(len(b))
		syscall.Fdatasync(int(f.Fd()))
		d := time.Since(t0)
		lat = append(lat, d)
		if d > 3*time.Millisecond {
			fmt.Printf("slow %v at +%v\n", d, t0.Sub(begin).Round(time.Millisecond))
		}
	}
	sort.Slice(lat, func(i, j int) bool { return lat[i] < lat[j] })
	fmt.Println("p50", lat[len(lat)/2], "p99", lat[len(lat)*99/100], "p999", lat[len(lat)*999/1000], "max", lat[len(lat)-1])
}
